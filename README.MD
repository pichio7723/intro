INSTALACIÓN DE PRISMA 7
npm install prisma@ --save-dev

INSTALAMOS EL CLIENTE PRISMA
npm install @prisma/client@7 --save-dev

INSTALAMOS EL ADAPTADOR
npm install @prisma/adapter-pg@7 --save-dev

INSTALAR LIBRERIA PARA ARCHIVOS EXTENCIÓN ts txt
npm install -D prisma@7 tsx

NOTA "--save-dev" ES IGUAL A "-D"

DESPUES DE INSTALAR LAS DEPENDENCIAS INICIAR PRISMA

npx prisma init --datasource-provider postgresql
